我找到了问题。JWT 过期时间在 src/auth/jwt.ts 被设为 5 分钟而非预期的 1 小时。以下是修复：

src/auth/jwt.ts
export function generateToken(payload: JwtPayload): string {
return jwt.sign(payload, SECRET, {
- expiresIn: '5m',
+ expiresIn: '1h',
});
}

已完成。我创建了支持轮换的新 refresh token 模块。

src/auth/refresh-token.ts 新文件
+import jwt from 'jsonwebtoken';
+import { Redis } from 'ioredis';
+
+const REFRESH_EXPIRY = '7d';
+const ROTATION_THRESHOLD = '1d';
+
+export async function rotateRefreshToken(
+ userId: string,
+ currentToken: string,
+ redis: Redis
+): Promise<string> {
+ // Revoke old token
+ await redis.del(`refresh:${currentToken}`);
+ // Issue new token
+ const newToken = jwt.sign({ userId, type: 'refresh' },
+ SECRET, { expiresIn: REFRESH_EXPIRY }
+ );
+ await redis.set(`refresh:${newToken}`, userId,
+ 'EX', 7 * 24 * 3600);
+ return newToken;
+}
